PSEUDOCODE OF ALPHA BETA PRUNING
-----------------------------------
ALPHA-BETA(node, depth, α, β, maximizingPlayer)

    if node is a terminal node OR depth = 0
        return utility value of node

    if maximizingPlayer = TRUE

        bestValue = -∞

        for each child of node

            value = ALPHA-BETA(
                        child,
                        depth - 1,
                        α,
                        β,
                        FALSE
                    )

            bestValue = MAX(bestValue, value)

            α = MAX(α, bestValue)

            if α ≥ β
                BREAK       // Prune remaining children

        return bestValue


    else

        bestValue = +∞

        for each child of node

            value = ALPHA-BETA(
                        child,
                        depth - 1,
                        α,
                        β,
                        TRUE
                    )

            bestValue = MIN(bestValue, value)

            β = MIN(β, bestValue)

            if α ≥ β
                BREAK       // Prune remaining children

        return bestValue
PYTHON CODE FOR ALPHA BETA PRUNING
----------------------------------------
def alpha_beta(node, depth, alpha, beta, maximizing_player):

    # Terminal node
    if depth == 0 or isinstance(node, int):
        return node

    if maximizing_player:

        best_value = float('-inf')

        for child in node:

            value = alpha_beta(
                child,
                depth - 1,
                alpha,
                beta,
                False
            )

            best_value = max(best_value, value)

            alpha = max(alpha, best_value)

            # Alpha-Beta pruning
            if alpha >= beta:
                break

        return best_value

    else:

        best_value = float('inf')

        for child in node:

            value = alpha_beta(
                child,
                depth - 1,
                alpha,
                beta,
                True
            )

            best_value = min(best_value, value)

            beta = min(beta, best_value)

            # Alpha-Beta pruning
            if alpha >= beta:
                break

        return best_value


# Game tree
tree = [
    [3, 5],
    [2, 9]
]

# Initial values
alpha = float('-inf')
beta = float('inf')

# MAX is the root
result = alpha_beta(
    tree,
    2,
    alpha,
    beta,
    True
)

print("Optimal value:", result)
